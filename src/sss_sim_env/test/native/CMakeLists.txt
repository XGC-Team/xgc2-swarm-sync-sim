cmake_minimum_required(VERSION 3.16)
project(SssClockNativeConformance LANGUAGES CXX)
find_package(XgcXrpc CONFIG REQUIRED)
find_package(Boost 1.75 REQUIRED COMPONENTS json)
find_package(Threads REQUIRED)
include(CTest)
add_executable(sss_clock_xrpc_test ../clock_xrpc_test.cpp ../../src/sim_clock/ClockControl.cpp)
target_include_directories(sss_clock_xrpc_test PRIVATE ../../include)
target_compile_features(sss_clock_xrpc_test PRIVATE cxx_std_20)
target_compile_options(sss_clock_xrpc_test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
target_link_libraries(sss_clock_xrpc_test PRIVATE XgcXrpc::http Boost::json Threads::Threads)
add_test(NAME sss_clock_native_xrpc COMMAND sss_clock_xrpc_test)
set_tests_properties(sss_clock_native_xrpc PROPERTIES TIMEOUT 30)
